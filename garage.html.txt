(function () {
  // Horaires : 0 = dimanche … 6 = samedi ; plages en minutes
  var PLAGES = {
    1: [[480,720],[840,1110]], 2: [[480,720],[840,1110]], 3: [[480,720],[840,1110]],
    4: [[480,720],[840,1110]], 5: [[480,720],[840,1110]], 6: [[480,720]], 0: []
  };
  var now = new Date();
  var jour = now.getDay(), min = now.getHours() * 60 + now.getMinutes();
  var ouvert = PLAGES[jour].some(function (p) { return min >= p[0] && min < p[1]; });
  var statut = document.getElementById('statut');
  statut.classList.toggle('ouvert', ouvert);
  statut.querySelector('span').textContent = ouvert ? 'Ouvert en ce moment' : 'Fermé actuellement — réservez en ligne';
  var ligne = document.querySelector('#table-horaires tr[data-j="' + jour + '"]');
  if (ligne) ligne.classList.add('aujourdhui');
  document.getElementById('annee').textContent = now.getFullYear();

  // Date : à partir de demain, max 3 mois
  var form = document.getElementById('form-rdv');
  var date = document.getElementById('date');
  function iso(d) { return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0'); }
  var demain = new Date(now); demain.setDate(now.getDate() + 1);
  var max = new Date(now); max.setMonth(now.getMonth() + 3);
  date.min = iso(demain); date.max = iso(max);

  var aprem = document.getElementById('lbl-aprem');
  var aprInput = aprem.querySelector('input');
  function jourChoisi() {
    if (!date.value) return null;
    var p = date.value.split('-');
    return new Date(+p[0], +p[1] - 1, +p[2]).getDay();
  }
  date.addEventListener('change', function () {
    var j = jourChoisi();
    var samedi = j === 6;
    aprInput.disabled = samedi;
    aprem.style.opacity = samedi ? '.4' : '1';
    if (samedi && aprInput.checked) aprInput.checked = false;
    validerDate();
  });

  function montrer(id, ok, msg) {
    var el = document.getElementById(id);
    if (msg) el.textContent = msg;
    el.style.display = ok ? 'none' : 'block';
    return ok;
  }
  function validerDate() {
    if (!date.value) return montrer('err-date', false, 'Choisissez une date.');
    if (date.value < date.min || date.value > date.max) return montrer('err-date', false, 'Choisissez une date entre demain et les 3 prochains mois.');
    if (jourChoisi() === 0) return montrer('err-date', false, 'Le garage est fermé le dimanche.');
    return montrer('err-date', true);
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var ok = true;
    ok = montrer('err-prestation', !!form.querySelector('input[name=prestation]:checked')) && ok;
    ok = validerDate() && ok;
    ok = montrer('err-creneau', !!form.querySelector('input[name=creneau]:checked')) && ok;
    ok = montrer('err-tel', /^[+\d][\d .\-]{8,}$/.test(form.telephone.value.trim())) && ok;
    ok = montrer('err-consent', form.consentement.checked) && ok;
    ['vehicule', 'nom', 'email'].forEach(function (n) {
      var el = form[n];
      var valide = el.checkValidity();
      el.style.borderColor = valide ? '' : 'var(--rouge)';
      ok = valide && ok;
    });
    if (!ok) {
      var premier = form.querySelector('.erreur[style*="block"], [style*="--rouge"]');
      if (premier) premier.scrollIntoView({ behavior: 'smooth', block: 'center' });
      return;
    }

    var btn = document.getElementById('btn-envoi');
    btn.disabled = true; btn.textContent = 'Envoi…';
    document.getElementById('err-envoi').style.display = 'none';
    var data = new FormData(form);
    if (data.get('immatriculation')) data.set('immatriculation', data.get('immatriculation').toUpperCase());

    fetch('/', {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams(data).toString()
    }).then(function (r) {
      if (!r.ok) throw new Error(r.status);
      var d = new Date(date.value + 'T12:00:00');
      var dateTxt = d.toLocaleDateString('fr-FR', { weekday: 'long', day: 'numeric', month: 'long' });
      var recap = document.getElementById('recap');
      recap.innerHTML = '';
      [['Prestation', data.get('prestation')], ['Véhicule', data.get('vehicule')], ['Date', dateTxt + ' — ' + data.get('creneau')]]
        .forEach(function (l) {
          var div = document.createElement('div');
          var b = document.createElement('strong'); b.textContent = l[0] + ' : ';
          div.appendChild(b); div.appendChild(document.createTextNode(l[1]));
          recap.appendChild(div);
        });
      form.style.display = 'none';
      document.getElementById('merci').style.display = 'block';
      document.getElementById('rdv').scrollIntoView({ behavior: 'smooth' });
    }).catch(function () {
      document.getElementById('err-envoi').style.display = 'block';
    }).finally(function () {
      btn.disabled = false; btn.textContent = 'Envoyer ma demande →';
    });
  });

  document.getElementById('nouvelle').addEventListener('click', function () {
    form.reset();
    aprInput.disabled = false; aprem.style.opacity = '1';
    form.style.display = 'block';
    document.getElementById('merci').style.display = 'none';
  });
})();
